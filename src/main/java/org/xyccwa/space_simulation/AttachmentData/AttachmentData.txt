package org.xyccwa.space_simulation.AttachmentData;

import com.mojang.serialization.Codec;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.attachment.AttachmentType;
import net.neoforged.neoforge.registries.DeferredRegister;
import net.neoforged.neoforge.registries.NeoForgeRegistries;
import org.xyccwa.space_simulation.SpaceSimulation;

import java.util.function.Supplier;

@EventBusSubscriber
public class AttachmentData {
    public static final DeferredRegister<AttachmentType<?>> ATTACHMENT_TYPES =
            DeferredRegister.create(NeoForgeRegistries.ATTACHMENT_TYPES, SpaceSimulation.MOD_ID);

    public static final Supplier<AttachmentType<Boolean>> is_wearing_spacesuit =
            ATTACHMENT_TYPES.register("is_wearing_spacesuit",
                    () -> AttachmentType.builder(() -> false)
                            .serialize(Codec.BOOL)
                            .build()
            );



    @SubscribeEvent
    public static void register() {

    }
}
